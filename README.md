 let type=filter; if(filter==='all') type=shuffle(['class','indications','contraindications','dose','routeText'])[0];
 let correct=norm(c[type]); let distract=shuffle(cards.filter(x=>x.name!==c.name && norm(x[type]) && norm(x[type])!==correct).map(x=>norm(x[type]))).slice(0,3);
 return {c,type,correct,options:shuffle([correct,...distract])};
}
